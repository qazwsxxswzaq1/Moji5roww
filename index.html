const ROWS = 7;
const COLS = 9;
const CONNECT = 5;

// عمق جستجو
const SEARCH_DEPTH = 4;

// امتیازهای برد
const WIN_SCORE = 1000000;

let gameBoard = [];
let currentPlayer = "red";
let gameOver = false;

const boardElement =
document.getElementById("board");

const scoresElement =
document.getElementById("scores");

const turnElement =
document.getElementById("turn");

const statusElement =
document.getElementById("status");


// ======================================
// شروع بازی
// ======================================

function startGame() {

gameBoard = [];

for (let row = 0; row < ROWS; row++) {

gameBoard[row] = [];

for (let col = 0; col < COLS; col++) {

gameBoard[row][col] = null;
}
}

currentPlayer = "red";
gameOver = false;

drawBoard();
calculateScores();
updateTurn();

statusElement.textContent =
"بازی شروع شد";
}


// ======================================
// رسم صفحه
// ======================================

function drawBoard() {

boardElement.innerHTML = "";

for (let row = 0; row < ROWS; row++) {

for (let col = 0; col < COLS; col++) {

const cell =
document.createElement("div");

cell.className = "cell";

if (gameBoard[row][col] === "red") {
cell.classList.add("red");
}

if (gameBoard[row][col] === "blue") {
cell.classList.add("blue");
}

cell.dataset.row = row;
cell.dataset.col = col;

cell.onclick = function() {
playColumn(col);
};

boardElement.appendChild(cell);
}
}
}


// ======================================
// حرکت دستی بازیکن
// ======================================

function playColumn(col) {

if (gameOver) {
return;
}

const row =
getAvailableRow(col);

if (row === -1) {

statusElement.textContent =
"⛔ این ستون پر است!";

return;
}

gameBoard[row][col] =
currentPlayer;

drawBoard();

const winningCells =
getWinningCells(
row,
col,
currentPlayer
);

if (winningCells.length >= CONNECT) {

gameOver = true;

statusElement.textContent =
currentPlayer === "red"
? "🏆 قرمز برنده شد!"
: "🏆 آبی برنده شد!";

highlightWinningCells(
winningCells
);

return;
}

if (isBoardFull()) {

gameOver = true;

statusElement.textContent =
"🤝 بازی مساوی شد!";

return;
}

currentPlayer =
currentPlayer === "red"
? "blue"
: "red";

updateTurn();
calculateScores();
}


// ======================================
// پیدا کردن پایین‌ترین خانه خالی
// ======================================

function getAvailableRow(col) {

for (
let row = ROWS - 1;
row >= 0;
row--
) {

if (gameBoard[row][col] === null) {
return row;
}
}

return -1;
}


// ======================================
// برد
// ======================================

function getWinningCells(
row,
col,
player
) {

const directions = [
[0, 1],
[1, 0],
[1, 1],
[1, -1]
];

for (const [dr, dc] of directions) {

let cells = [[row, col]];

let r = row + dr;
let c = col + dc;

while (
r >= 0 &&
r < ROWS &&
c >= 0 &&
c < COLS &&
gameBoard[r][c] === player
) {

cells.push([r, c]);

r += dr;
c += dc;
}

r = row - dr;
c = col - dc;

while (
r >= 0 &&
r < ROWS &&
c >= 0 &&
c < COLS &&
gameBoard[r][c] === player
) {

cells.push([r, c]);

r -= dr;
c -= dc;
}

if (cells.length >= CONNECT) {
return cells;
}
}

return [];
}


// ======================================
// بررسی اینکه بازیکن برنده شده یا نه
// ======================================

function hasWinner(player) {

for (let row = 0; row < ROWS; row++) {

for (let col = 0; col < COLS; col++) {

if (gameBoard[row][col] !== player) {
continue;
}

if (
getWinningCells(
row,
col,
player
).length >= CONNECT
) {
return true;
}
}
}

return false;
}


// ======================================
// ارزیابی یک خط ۵ خانه‌ای
// ======================================

function scoreWindow(
window,
player,
opponent
) {

let playerCount = 0;
let opponentCount = 0;
let emptyCount = 0;

for (const cell of window) {

if (cell === player) {
playerCount++;
}

else if (cell === opponent) {
opponentCount++;
}

else {
emptyCount++;
}
}

let score = 0;


// برد کامل
if (playerCount === 5) {
score += 100000;
}

// چهار مهره + جای خالی
else if (
playerCount === 4 &&
emptyCount === 1
) {
score += 5000;
}

// سه مهره + دو جای خالی
else if (
playerCount === 3 &&
emptyCount === 2
) {
score += 400;
}

// دو مهره
else if (
playerCount === 2 &&
emptyCount === 3
) {
score += 40;
}

// یک مهره
else if (
playerCount === 1 &&
emptyCount === 4
) {
score += 5;
}


// تهدید حریف
if (
opponentCount === 4 &&
emptyCount === 1
) {
score -= 7000;
}

else if (
opponentCount === 3 &&
emptyCount === 2
) {
score -= 500;
}

else if (
opponentCount === 2 &&
emptyCount === 3
) {
score -= 40;
}


return score;
}


// ======================================
// ارزیابی کامل صفحه
// ======================================

function evaluateBoard(player) {

const opponent =
player === "red"
? "blue"
: "red";

if (hasWinner(player)) {
return WIN_SCORE;
}

if (hasWinner(opponent)) {
return -WIN_SCORE;
}


let score = 0;


// مرکز صفحه
const center =
Math.floor(COLS / 2);

for (let row = 0; row < ROWS; row++) {

if (
gameBoard[row][center] === player
) {
score += 12;
}

if (
gameBoard[row][center] === opponent
) {
score -= 12;
}
}


const directions = [
[0, 1],
[1, 0],
[1, 1],
[1, -1]
];


for (let row = 0; row < ROWS; row++) {

for (let col = 0; col < COLS; col++) {

for (const [dr, dc] of directions) {

let window = [];
let valid = true;


for (
let i = 0;
i < CONNECT;
i++
) {

const r =
row + dr * i;

const c =
col + dc * i;


if (
r < 0 ||
r >= ROWS ||
c < 0 ||
c >= COLS
) {

valid = false;
break;
}


window.push(
gameBoard[r][c]
);
}


if (!valid) {
continue;
}


score += scoreWindow(
window,
player,
opponent
);
}
}
}


return score;
}


// ======================================
// Minimax + Alpha Beta
// ======================================

function minimax(
depth,
maximizing,
aiPlayer,
alpha,
beta
) {

const opponent =
aiPlayer === "red"
? "blue"
: "red";


// وضعیت نهایی
if (hasWinner(aiPlayer)) {
return WIN_SCORE + depth;
}

if (hasWinner(opponent)) {
return -WIN_SCORE - depth;
}

if (depth === 0) {
return evaluateBoard(aiPlayer);
}

if (isBoardFull()) {
return 0;
}


// نوبت AI
if (maximizing) {

let bestScore = -Infinity;


for (
const col of getOrderedColumns()
) {

const row =
getAvailableRow(col);

if (row === -1) {
continue;
}


gameBoard[row][col] =
aiPlayer;


const score =
minimax(
depth - 1,
false,
aiPlayer,
alpha,
beta
);


gameBoard[row][col] =
null;


bestScore =
Math.max(
bestScore,
score
);


alpha =
Math.max(
alpha,
bestScore
);


// Alpha-Beta pruning
if (beta <= alpha) {
break;
}
}


return bestScore;
}


// نوبت حریف
else {

let bestScore = Infinity;


for (
const col of getOrderedColumns()
) {

const row =
getAvailableRow(col);

if (row === -1) {
continue;
}


gameBoard[row][col] =
opponent;


const score =
minimax(
depth - 1,
true,
aiPlayer,
alpha,
beta
);


gameBoard[row][col] =
null;


bestScore =
Math.min(
bestScore,
score
);


beta =
Math.min(
beta,
bestScore
);


// Alpha-Beta pruning
if (beta <= alpha) {
break;
}
}


return bestScore;
}
}


// ======================================
// ترتیب بررسی ستون‌ها
// مرکز اول
// ======================================

function getOrderedColumns() {

const center =
Math.floor(COLS / 2);

const columns = [];

for (
let distance = 0;
distance <= center;
distance++
) {

const left =
center - distance;

const right =
center + distance;


if (
left >= 0 &&
!columns.includes(left)
) {
columns.push(left);
}


if (
right < COLS &&
!columns.includes(right)
) {
columns.push(right);
}
}

return columns;
}


// ======================================
// پیدا کردن بهترین حرکت
// ======================================

function getBestMove() {

let bestColumn = -1;
let bestScore = -Infinity;


// اول: برد فوری
for (
let col = 0;
col < COLS;
col++
) {

const row =
getAvailableRow(col);

if (row === -1) {
continue;
}


gameBoard[row][col] =
currentPlayer;


const win =
hasWinner(currentPlayer);


gameBoard[row][col] =
null;


if (win) {

return {
column: col,
score: WIN_SCORE
};
}
}


// دوم: Minimax
for (
const col of getOrderedColumns()
) {

const row =
getAvailableRow(col);

if (row === -1) {
continue;
}


gameBoard[row][col] =
currentPlayer;


const score =
minimax(
SEARCH_DEPTH - 1,
false,
currentPlayer,
-Infinity,
Infinity
);


gameBoard[row][col] =
null;


if (score > bestScore) {

bestScore = score;
bestColumn = col;
}
}


return {
column: bestColumn,
score: bestScore
};
}


// ======================================
// امتیاز ستون‌ها
// ======================================

function calculateScores() {

scoresElement.innerHTML = "";


let scores = [];


for (
let col = 0;
col < COLS;
col++
) {

const row =
getAvailableRow(col);


if (row === -1) {

scores.push("-");

continue;
}


gameBoard[row][col] =
currentPlayer;


const score =
minimax(
SEARCH_DEPTH - 1,
false,
currentPlayer,
-Infinity,
Infinity
);


gameBoard[row][col] =
null;


scores.push(score);
}


const validScores =
scores.filter(
score => score !== "-"
);


const bestScore =
Math.max(
...validScores
);


for (
let col = 0;
col < COLS;
col++
) {

const score =
document.createElement("div");

score.className =
"score";


score.textContent =
scores[col];


if (
scores[col] === bestScore
) {

score.classList.add("best");
}


scoresElement.appendChild(
score
);
}
}


// ======================================
// Hint
// ======================================

function showHint() {

if (gameOver) {
return;
}


statusElement.textContent =
"🧠 در حال بررسی ۴ حرکت جلوتر...";


setTimeout(function() {

const result =
getBestMove();


if (result.column === -1) {

statusElement.textContent =
"حرکت ممکنی وجود ندارد.";

return;
}


statusElement.textContent =
"🧠 پیشنهاد: ستون " +
(result.column + 1) +
" | امتیاز: " +
result.score;


// فقط نمایش خانه پیشنهادی
clearHint();


const row =
getAvailableRow(
result.column
);


if (row !== -1) {

const cell =
document.querySelector(
`[data-row="${row}"][data-col="${result.column}"]`
);


if (cell) {
cell.classList.add(
"hint-cell"
);
}
}

}, 50);
}


// ======================================
// پاک کردن Hint
// ======================================

function clearHint() {

document
.querySelectorAll(".hint-cell")
.forEach(function(cell) {

cell.classList.remove(
"hint-cell"
);

});
}


// ======================================
// صفحه پر است
// ======================================

function isBoardFull() {

for (
let col = 0;
col < COLS;
col++
) {

if (
gameBoard[0][col] === null
) {
return false;
}
}

return true;
}


// ======================================
// نوبت
// ======================================

function updateTurn() {

if (gameOver) {
return;
}


if (currentPlayer === "red") {

turnElement.textContent =
"نوبت: 🔴 قرمز";

} else {

turnElement.textContent =
"نوبت: 🔵 آبی";
}
}


// ======================================
// Restart
// ======================================

function restartGame() {

}

startGame();
